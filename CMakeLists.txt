cmake_minimum_required(VERSION 3.25.0)

#------------------------------------------------------------------------
# KT Delay - VST 3 plug-in (VST 3 SDK + VSTGUI, no JUCE)
#
#   cmake -G "Visual Studio 17 2022" -A x64 -S . -B build
#   cmake --build build --config Release
#   -> build/VST3/Release/KTDelay.vst3
#------------------------------------------------------------------------

# SDK options (must be set before add_subdirectory of the SDK)
option(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES "Enable VST 3 Plug-in Examples" OFF)
option(SMTG_ENABLE_VST3_HOSTING_EXAMPLES "Enable VST 3 Hosting Examples" OFF)
option(SMTG_ENABLE_VSTGUI_SUPPORT "Enable VSTGUI Support" ON)
# creating the symlink into the system VST3 folder needs admin rights on Windows
option(SMTG_CREATE_PLUGIN_LINK "Create symbolic link for each Plug-in" OFF)

set(CMAKE_OSX_DEPLOYMENT_TARGET 10.13 CACHE STRING "")

# Location of the VST 3 SDK ("vst3sdk" folder). Not part of this repository.
#   -Dvst3sdk_SOURCE_DIR=<path>   or   environment variable VST3_SDK_ROOT
#   default: <repo>/VST_SDK/vst3sdk
set(vst3sdk_SOURCE_DIR "" CACHE PATH "Path to the vst3sdk folder of the Steinberg VST 3 SDK")
if(NOT vst3sdk_SOURCE_DIR)
    if(DEFINED ENV{VST3_SDK_ROOT})
        set(vst3sdk_SOURCE_DIR "$ENV{VST3_SDK_ROOT}")
    else()
        set(vst3sdk_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/VST_SDK/vst3sdk")
    endif()
endif()
if(NOT EXISTS "${vst3sdk_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR
        "VST 3 SDK not found at '${vst3sdk_SOURCE_DIR}'.\n"
        "Download it from https://www.steinberg.net/developers/ and either extract it to "
        "<repo>/VST_SDK/vst3sdk or pass -Dvst3sdk_SOURCE_DIR=<path> (see docs/DEVELOPMENT.md).")
endif()

project(KTDelay
    # Plug-in version. Symbols are generated into projectversion.h (included by source/version.h).
    VERSION 1.0.0.0
    DESCRIPTION "KT Delay VST 3 Plug-in"
)

set(SMTG_VSTGUI_ROOT "${vst3sdk_SOURCE_DIR}")

add_subdirectory(${vst3sdk_SOURCE_DIR} ${PROJECT_BINARY_DIR}/vst3sdk)
smtg_enable_vst3_sdk()

smtg_add_vst3plugin(KTDelay
    source/version.h
    source/ktdelaycids.h
    source/ktdelayprocessor.h
    source/ktdelayprocessor.cpp
    source/ktdelaycontroller.h
    source/ktdelaycontroller.cpp
    source/ktdelayeditor.h
    source/ktdelayeditor.cpp
    source/ktdelayentry.cpp
)

target_compile_features(KTDelay PUBLIC cxx_std_17)

if(NOT SMTG_ENABLE_VSTGUI_SUPPORT)
    message(FATAL_ERROR "KT Delay needs VSTGUI (SMTG_ENABLE_VSTGUI_SUPPORT=ON)")
endif()

target_link_libraries(KTDelay
    PRIVATE
        sdk
        vstgui_support
)

# GUI images (generated by tools/gen_images.py) -> KTDelay.vst3/Contents/Resources/
smtg_target_add_plugin_resources(KTDelay
    RESOURCES
        resource/background.png
        resource/knob.png
        resource/bypass.png
)

smtg_target_add_plugin_snapshots(KTDelay
    RESOURCES
        resource/E4B7EC890B0F489DB6EE9737C605845F_snapshot.png
)

smtg_target_configure_version_file(KTDelay)

#------------------------------------------------------------------------
# optional offline DSP test (no host needed)
#   cmake -S . -B build -DKTDELAY_BUILD_TESTS=ON
#   cmake --build build --config Release --target ktdelay_dsptest
#------------------------------------------------------------------------
option(KTDELAY_BUILD_TESTS "Build the KT Delay offline DSP test" OFF)
if(KTDELAY_BUILD_TESTS)
    add_executable(ktdelay_dsptest
        tests/dsptest.cpp
        source/ktdelayprocessor.cpp
        source/ktdelayprocessor.h
        source/ktdelaycids.h
        ${vst3sdk_SOURCE_DIR}/public.sdk/source/common/memorystream.cpp
    )
    target_include_directories(ktdelay_dsptest PRIVATE source)
    target_compile_features(ktdelay_dsptest PUBLIC cxx_std_17)
    target_link_libraries(ktdelay_dsptest PRIVATE sdk sdk_common base pluginterfaces)
    enable_testing()
    add_test(NAME ktdelay_dsptest COMMAND ktdelay_dsptest)
endif()

if(SMTG_MAC)
    smtg_target_set_bundle(KTDelay
        BUNDLE_IDENTIFIER com.aikelab.vst3.ktdelay
        COMPANY_NAME "aikelab"
    )
elseif(SMTG_WIN)
    target_sources(KTDelay PRIVATE
        resource/win32resource.rc
    )
    if(MSVC)
        set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT KTDelay)
    endif()
endif()
